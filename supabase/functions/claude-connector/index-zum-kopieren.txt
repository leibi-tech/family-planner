function isoWeek(d) {
  const x = new Date(Date.UTC(d.getFullYear(), d.getMonth(), d.getDate()));
  const day = x.getUTCDay() || 7;
  x.setUTCDate(x.getUTCDate() + 4 - day);
  const yearStart = new Date(Date.UTC(x.getUTCFullYear(), 0, 1));
  return Math.ceil(((x.getTime() - yearStart.getTime()) / 864e5 + 1) / 7);
}

function roundQty(qty, unit) {
  const up = (x, step) => Math.ceil(x / step - 1e-9) * step;
  let r;
  switch (unit) {
    case "g":
    case "ml":
      r = qty < 10 ? up(qty, 1) : up(qty, 10);
      break;
    case "kg":
    case "l":
      r = up(qty, 0.1);
      break;
    case "EL":
    case "TL":
      r = up(qty, 0.5);
      break;
    default:
      r = up(qty, 1);
  }
  return Math.round(r * 100) / 100;
}
function buildShoppingList(monday, dayMap, menuMap, persons, extras = []) {
  const acc = /* @__PURE__ */ new Map();
  const getItem = (product, unit, category) => {
    const k = itemKey(product, unit);
    let item = acc.get(k);
    if (!item) {
      const name = product.trim();
      item = { key: k, product: name.charAt(0).toUpperCase() + name.slice(1), unit, qty: null, raw: 0, category, sources: [] };
      acc.set(k, item);
    }
    if (category === "lebensmittel") item.category = "lebensmittel";
    return item;
  };
  for (let i = 0; i < 7; i++) {
    const key = dateKey(addDays(monday, i));
    const day = dayMap.get(key);
    const slots = day ? day.slots : defaultSlots(key);
    for (const slot of slots) {
      if (!slot.active) continue;
      const eaters = persons.filter((p) => !slot.absent.includes(p.id));
      const factor = eaters.reduce((s, p) => s + p.factor, 0);
      if (factor <= 0) continue;
      for (const dish of slot.dishes) {
        if (dish.shop === false) continue;
        const menu = menuMap.get(dish.menuId);
        if (!menu) continue;
        const source = { day: WEEKDAYS[i].slice(0, 2), slot: slot.label, menu: menu.name, kind: "plan" };
        for (const ing of menu.ingredients) {
          const item = getItem(ing.product, ing.unit, ing.category);
          if (ing.amount != null && menu.basis.baseFactor > 0) {
            item.raw += ing.amount / menu.basis.baseFactor * factor;
          }
          const last = item.sources[item.sources.length - 1];
          if (!last || last.day !== source.day || last.slot !== source.slot || last.menu !== source.menu) {
            item.sources.push(source);
          }
        }
      }
    }
  }
  for (const x of extras) {
    if (!x.product.trim()) continue;
    const item = getItem(x.product, x.unit, x.category);
    if (x.qty != null) item.raw += x.qty;
    item.sources.push(x.source);
  }
  return [...acc.values()].map(({ raw, ...item }) => ({
    ...item,
    qty: raw > 0 ? roundQty(raw, item.unit) : null
  }));
}

async function toolEventUpdate(ctx, a) {

  const r = await loadEvents(ctx, dateKey(addDays(parseKey(day), -1)), dateKey(addDays(parseKey(day), 2)));

  let endDate = isDateKey(a?.bis_datum) ? a.bis_datum : dateKey(addDays(parseKey(oldEndDate), shift));
  let allDay = e.allDay;
  let startTime = e.allDay ? void 0 : hhmm(e.start);
  let endTime = e.allDay ? void 0 : hhmm(e.end);
  if (a?.ganztaegig === true) {
    allDay = true;
    startTime = endTime = void 0;
  } else if (isTime(a?.start)) {
    const dur = e.allDay ? 60 : Math.max(15, Math.round((+new Date(e.end) - +new Date(e.start)) / 6e4));
    allDay = false;
    startTime = a.start;
    if (isTime(a?.ende)) {
      endTime = a.ende;
      if (!isDateKey(a?.bis_datum)) endDate = endTime <= startTime ? dateKey(addDays(parseKey(startDate), 1)) : startDate;
    } else [endDate, endTime] = addMinutes(startDate, startTime, dur);
  } else if (isTime(a?.ende) && !allDay) {
    endTime = a.ende;
  }
  const parsed = parseTitle(e.title, ctx.persons);
  const newTitle = a?.titel !== void 0 ? String(a.titel).trim() : parsed.title;
  if (!newTitle) throw new HttpError(400, "Titel darf nicht leer sein.");
  let full = e.title;
  let inv = {};
  if (a?.personen !== void 0) {
    const who = whoParts(ctx, cal, r.calendars, newTitle, a.personen);
    full = who.full;
    if (who.invite) inv = { invite: who.invite, inviteKnown: who.inviteKnown };
  } else if (a?.titel !== void 0) {
    full = buildTitle(newTitle, parsed.personIds, ctx.persons, parsed.only);
  }
  const event = {
    title: full,
    allDay,
    startDate,
    endDate,
    startTime,
    endTime,
    location: a?.ort !== void 0 ? String(a.ort ?? "") : e.location ?? "",
    description: a?.notiz !== void 0 ? String(a.notiz ?? "") : e.description ?? ""
  };
  await calCall(ctx, "update", { calendarId: calId, eventId: id, recurringEventId: e.recurringEventId, scope: "one", event, ...inv });
  return `Termin ge\xE4ndert: ${full}, ${allDay ? fmtDate(startDate) + " ganzt\xE4gig" : `${fmtDate(startDate)} ${startTime}\u2013${endTime}`}${e.recurringEventId ? " (nur dieser eine Termin der Serie)" : ""}.`;
}

async function actMcp(req) {
  if (req.method === "GET") return json({ error: "Nur POST (kein Ereignis-Strom)." }, 405, { Allow: "POST" });
  if (req.method !== "POST") return json({ error: "Nicht erlaubt." }, 405, { Allow: "POST" });
  const auth = req.headers.get("authorization") ?? "";
  const tok = auth.startsWith("Bearer ") ? await unsign(auth.slice(7)) : null;
  if (!tok || tok.k !== "a") return unauthorized("Bitte mit myFam. verbinden.");
  if (tok.e < Date.now()) return unauthorized("Zugang abgelaufen.");
  const conn = (await adminDb(`claude_connections?id=eq.${tok.s}&select=*`))?.[0];
  if (!conn || !conn.refresh_hash) return unauthorized("Verbindung getrennt. Bitte neu verbinden.");
  const msg = await req.json().catch(() => null);
  if (!msg) return json({ jsonrpc: "2.0", id: null, error: { code: -32700, message: "Ung\xFCltiges JSON." } }, 400);
  const batch = Array.isArray(msg);
  const msgs = batch ? msg : [msg];
  let ctx = null;
  const getCtx = async () => {
    if (ctx) return ctx;
    const token = await sbAccess(conn);
    if (!token) {
      await adminDb(`claude_connections?id=eq.${conn.id}`, { method: "DELETE" });
      throw new HttpError(401, "Die Anmeldung bei myFam. ist abgelaufen. Bitte den Connector neu verbinden.");
    }
    ctx = await makeCtx(conn, token);
    return ctx;
  };
  const replies = [];
  for (const m of msgs) {
    if (!m || m.jsonrpc !== "2.0" || typeof m.method !== "string") {
      if (m && m.id !== void 0 && (m.result !== void 0 || m.error !== void 0)) continue;
      replies.push({ jsonrpc: "2.0", id: m?.id ?? null, error: { code: -32600, message: "Ung\xFCltige Anfrage." } });
      continue;
    }
    const isNote = m.id === void 0 || m.id === null;
    const ok = (result) => !isNote && replies.push({ jsonrpc: "2.0", id: m.id, result });
    const err = (code, message) => !isNote && replies.push({ jsonrpc: "2.0", id: m.id, error: { code, message } });
    try {
      switch (m.method) {
        case "initialize": {
          const want = String(m.params?.protocolVersion ?? "");
          ok({
            protocolVersion: PROTOCOLS.includes(want) ? want : PROTOCOLS[1],
            capabilities: { tools: { listChanged: false } },
            serverInfo: { name: "myfam", title: "myFam.", version: VERSION },
            instructions: INSTRUCTIONS
          });
          break;
        }
        case "ping":
          ok({});
          break;
        case "tools/list":
          ok({ tools: TOOLS });
          break;
        case "tools/call": {
          const name = String(m.params?.name ?? "");
          const h = HANDLERS[name];
          if (!h) {
            err(-32602, `Unbekanntes Werkzeug: ${name}`);
            break;
          }
          try {
            const text = await h(await getCtx(), m.params?.arguments ?? {});
            ok({ content: [{ type: "text", text }], isError: false });
          } catch (e) {
            if (e instanceof HttpError && e.status === 401) return unauthorized(e.message);
            const text = e instanceof HttpError ? e.message : "Unerwarteter Fehler im myFam.-Connector.";
            if (!(e instanceof HttpError)) console.error(e);
            ok({ content: [{ type: "text", text }], isError: true });
          }
          break;
        }
        default:
          if (m.method.startsWith("notifications/")) break;
          err(-32601, `Unbekannte Methode: ${m.method}`);
      }
    } catch (e) {
      err(-32603, e instanceof HttpError ? e.message : "Interner Fehler.");
    }
  }
  if (!replies.length) return new Response(null, { status: 202, headers: CORS });
  return json(batch ? replies : replies[0]);
}
function subPath(pathname) {
  const i = pathname.indexOf("/" + FN);
  let p = i >= 0 ? pathname.slice(i + FN.length + 1) : pathname;
  p = p.replace(/\/+$/, "");
  return p || "/";
}
async function handler(req) {
  if (req.method === "OPTIONS") return new Response("ok", { headers: CORS });
  const url = new URL(req.url);
  const p = subPath(url.pathname);
  try {
    if (p.startsWith("/.well-known/oauth-protected-resource")) return json(resourceMetadata());
    if (p.startsWith("/.well-known/oauth-authorization-server") || p.startsWith("/.well-known/openid-configuration")) return json(serverMetadata());
    if (p === "/register" && req.method === "POST") return await actRegister(req);
    if (p === "/authorize" && req.method === "GET") return await actAuthorize(url);
    if (p === "/approve" && req.method === "POST") return await actApprove(req);
    if (p === "/token" && req.method === "POST") return await actToken(req);
    if (p === "/" || p === "/mcp") return await actMcp(req);
    return json({ error: "Nicht gefunden." }, 404);
  } catch (e) {
    if (e instanceof HttpError) return json({ error: e.message }, e.status);
    console.error(e);
    return json({ error: "Unerwarteter Fehler im myFam.-Connector." }, 500);
  }
}
